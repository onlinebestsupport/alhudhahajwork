def _app_base():
    if getattr(sys, "frozen", False):
        return os.path.dirname(sys.executable)
    return os.path.dirname(os.path.abspath(__file__))


def _fmt_inr(value):
    try:
        v = float(value or 0)
    except (TypeError, ValueError):
        return "0.00"
    if v != v:
        return "0.00"
    negative = v < 0
    v = abs(v)
    v = round(v, 2)
    int_part = int(v)
    dec_part = int(round((v - int_part) * 100))
    if dec_part >= 100:
        int_part += 1
        dec_part = 0
    s = str(int_part)
    if len(s) <= 3:
        grouped = s
    else:
        last3 = s[-3:]
        rest = s[:-3]
        groups = []
        while len(rest) > 2:
            groups.insert(0, rest[-2:])
            rest = rest[:-2]
        if rest:
            groups.insert(0, rest)
        grouped = ",".join(groups) + "," + last3
    out = f"{grouped}.{dec_part:02d}"
    return f"-{out}" if negative else out


def _fmt_date_ddmmyyyy(date_val):
    if date_val is None:
        return ""
    s = str(date_val).strip()
    if not s or s.lower() in ("nan", "none", "nat"):
        return ""
    if len(s) >= 10 and s[4] == "-" and s[7] == "-":
        return f"{s[8:10]}/{s[5:7]}/{s[0:4]}"
    for fmt in ("%Y-%m-%dT%H:%M:%S", "%Y-%m-%d %H:%M:%S",
                "%Y-%m-%d", "%d-%m-%Y", "%d/%m/%Y"):
        try:
            return datetime.strptime(s[:19], fmt).strftime("%d/%m/%Y")
        except Exception:
            continue
    return s[:10]

    async def _refresh_hub_stats_async(self):
        import asyncio
        await asyncio.sleep(0.1)
        self._refresh_hub_stats()

    def _refresh_hub_stats(self, e=None):
        try:
            travelers = self.db.get_travelers()
            batches = self.db.get_batches()
            payments = self.db.get_payments()
            invoices = self.db.get_invoices()

            n_travelers = len(travelers)
            n_batches = len(batches)
            n_payments = len(payments)
            n_invoices = len(invoices)

            total_revenue = 0.0
            for inv in invoices:
                try:
                    total_revenue += float(inv.get("total_amount", 0) or 0)
                except (TypeError, ValueError):
                    pass

            base = _app_base()
            n_photos = 0
            for t in travelers:
                p = (t.get("photo", "") or "").strip()
                if p and os.path.exists(os.path.join(base, p)):
                    n_photos += 1
                    continue
                tid = t.get("id", "")
                if tid:
                    folder = tid.replace("/", "_").replace("\\", "_")
                    pdir = os.path.join(base, "documents", folder, "photos")
                    if os.path.exists(pdir):
                        for f in os.listdir(pdir):
                            if f.lower().endswith((".jpg", ".jpeg", ".png",
                                                    ".bmp", ".gif")):
                                n_photos += 1
                                break

            if "travelers" in self._hub_stats:
                self._hub_stats["travelers"].value = str(n_travelers)
            if "batches" in self._hub_stats:
                self._hub_stats["batches"].value = str(n_batches)
            if "payments" in self._hub_stats:
                self._hub_stats["payments"].value = str(n_payments)
            if "invoices" in self._hub_stats:
                self._hub_stats["invoices"].value = str(n_invoices)
            if "revenue" in self._hub_stats:
                if total_revenue >= 10000000:
                    rev = f"₹{total_revenue/10000000:.1f}Cr"
                elif total_revenue >= 100000:
                    rev = f"₹{total_revenue/100000:.1f}L"
                elif total_revenue >= 1000:
                    rev = f"₹{total_revenue/1000:.1f}K"
                else:
                    rev = f"₹{_fmt_inr(total_revenue)}"
                self._hub_stats["revenue"].value = rev

            if "custom" in self._hub_cards:
                self._hub_cards["custom"].value = "📊  39 fields available"
            if "single" in self._hub_cards:
                self._hub_cards["single"].value = (
                    f"👥  {n_travelers} travelers ready")
            if "standard" in self._hub_cards:
                self._hub_cards["standard"].value = (
                    "📋  4 report types available")
            if "photo" in self._hub_cards:
                self._hub_cards["photo"].value = (
                    f"📷  {n_photos} photos available")

            try:
                self.page.update()
            except Exception:
                pass
        except Exception as e:
            print(f"_refresh_hub_stats: {e}")

    def _on_single_search(self, e=None):
        try:
            self._single_result.controls.clear()
            q = (self._single_search.value or "").strip().lower()
            if not q:
                self._single_result.controls.append(
                    ft.Text("Enter a search query.",
                            color=ft.Colors.GREY_500))
                self.page.update()
                return

            travelers = self.db.get_travelers()
            matches = []
            for t in travelers:
                haystack = (
                    f"{t.get('first_name','')} {t.get('last_name','')} "
                    f"{t.get('passport_no','')} {t.get('id','')} "
                    f"{t.get('file_reference','')}"
                ).lower()
                if q in haystack:
                    matches.append(t)

            if not matches:
                self._single_result.controls.append(
                    ft.Text("No travelers matched.",
                            color=ft.Colors.GREY_500))
                self.page.update()
                return

            for t in matches[:20]:
                self._single_result.controls.append(
                    self._make_traveler_card(t))
            self.page.update()
        except Exception as ex:
            print(f"_on_single_search: {ex}")

    def _open_traveler_report(self, t):
        tid = t.get("id", "")
        batch = (self.db.get_batch_by_id(t.get("batch_id"))
                 if t.get("batch_id") else None)
        payments = self.db.get_payments(tid) if tid else []
        invoices = self.db.get_invoices(tid) if tid else []

        total_paid = sum(float(p.get("amount", 0) or 0) for p in payments)
        batch_price = float(batch.get("price", 0) or 0) if batch else 0.0
        pending = max(0.0, batch_price - total_paid)

        def _row(label, value):
            return ft.Row([
                ft.Text(label, width=180, size=12,
                        weight=ft.FontWeight.BOLD,
                        color=ft.Colors.GREY_700),
                ft.Text(str(value) if value not in (None, "") else "—",
                        size=12, selectable=True, expand=True),
            ], spacing=8)

        dialog = ft.AlertDialog(
            modal=True,
            title=ft.Text(
                f"📋  {(t.get('first_name') or '')} "
                f"{(t.get('last_name') or '')}",
                weight=ft.FontWeight.BOLD),
            content=ft.Container(
                content=ft.Column([
                    _row("Traveler ID", t.get("id")),
                    _row("File Reference", t.get("file_reference")),
                    _row("Passport", t.get("passport_no")),
                    _row("Mobile", t.get("mobile")),
                    _row("Email", t.get("email")),
                    _row("Status", t.get("status")),
                    ft.Divider(),
                    _row("Batch",
                         batch.get("batch_name") if batch else "—"),
                    _row("Package Price", f"₹{_fmt_inr(batch_price)}"),
                    ft.Divider(),
                    _row("Payments", len(payments)),
                    _row("Total Paid", f"₹{_fmt_inr(total_paid)}"),
                    _row("Pending", f"₹{_fmt_inr(pending)}"),
                    _row("Invoices", len(invoices)),
                ], spacing=6, scroll=ft.ScrollMode.AUTO),
                width=500, height=500, padding=10,
            ),
            actions=[
                ft.TextButton(
                    content=ft.Text("Close"),
                    on_click=lambda e: self.page.pop_dialog()),
            ],
        )

    def _generate_standard(self, e=None):
        try:
            sd = self._std_start.value.strip()
            ed = self._std_end.value.strip()
            rt = self._std_report_type.value

            if rt == "Financial Summary":
                txt = self._report_financial(sd, ed)
            elif rt == "Traveler Statistics":
                txt = self._report_travelers(sd, ed)
            elif rt == "Batch Utilization":
                txt = self._report_batches(sd, ed)
            else:
                txt = self._report_payments(sd, ed)

            self._std_display.value = txt
            self.page.update()
        except Exception as ex:
            self._std_display.value = f"❌ Error: {ex}"
            self.page.update()

    def _report_financial(self, sd, ed):
        payments = self.db.get_payments()
        filtered = []
        for p in payments:
            pd_ = str(p.get("payment_date", ""))[:10]
            if sd <= pd_ <= ed:
                filtered.append(p)

        total = sum(float(p.get("amount", 0) or 0) for p in filtered)
        by_method = {}
        for p in filtered:
            m = p.get("payment_method", "Other")
            by_method[m] = (by_method.get(m, 0)
                            + float(p.get("amount", 0) or 0))

        lines = [f"FINANCIAL SUMMARY — {sd} to {ed}", "=" * 60, ""]
        lines.append(f"Total Collections : ₹{_fmt_inr(total)}")
        lines.append(f"Number of Payments: {len(filtered)}")
        avg = total / len(filtered) if filtered else 0
        lines.append(f"Average Payment   : ₹{_fmt_inr(avg)}")
        lines.append("")
        lines.append("Payment by Method:")
        for m, amt in by_method.items():
            lines.append(f"  • {m}: ₹{_fmt_inr(amt)}")
        return "\n".join(lines)

    def _report_travelers(self, sd, ed):
        travelers = self.db.get_travelers()
        filtered = []
        for t in travelers:
            rdate = str(t.get("registration_date", ""))[:10]
            if not rdate or (sd <= rdate <= ed):
                filtered.append(t)
        status_count = {}
        for t in filtered:
            s = t.get("status", "Unknown")
            status_count[s] = status_count.get(s, 0) + 1
        lines = [f"TRAVELER STATISTICS — {sd} to {ed}", "=" * 60, ""]
        lines.append(f"Total Travelers: {len(filtered)}")
        lines.append("")
        lines.append("Status Distribution:")
        for s, c in status_count.items():
            p = (c / len(filtered) * 100) if filtered else 0
            lines.append(f"  • {s}: {c} ({p:.1f}%)")
        return "\n".join(lines)

    def _report_batches(self, sd, ed):
        batches = self.db.get_batches()
        travelers = self.db.get_travelers()
        counts = {}
        for t in travelers:
            bid = t.get("batch_id")
            if bid:
                counts[bid] = counts.get(bid, 0) + 1

        lines = [f"BATCH UTILIZATION — {sd} to {ed}", "=" * 60, ""]
        total_seats = 0
        total_booked = 0
        total_rev = 0.0
        for b in batches:
            bid = b.get("id")
            booked = counts.get(bid, 0)
            total = int(b.get("total_seats", 0) or 0)
            price = float(b.get("price", 0) or 0)
            rev = booked * price
            total_seats += total
            total_booked += booked
            total_rev += rev
            pct = (booked / total * 100) if total > 0 else 0
            lines.append(
                f"📌 {b.get('batch_name','Unknown')}: "
                f"{booked}/{total} ({pct:.1f}%) — ₹{_fmt_inr(rev)}")
        lines.append("")
        lines.append("=" * 60)
        lines.append(f"Total Seats  : {total_seats}")
        lines.append(f"Total Booked : {total_booked}")
        if total_seats > 0:
            lines.append(
                f"Occupancy    : {total_booked/total_seats*100:.1f}%")
        lines.append(f"Total Revenue: ₹{_fmt_inr(total_rev)}")
        return "\n".join(lines)

    def _report_payments(self, sd, ed):
        payments = self.db.get_payments()
        filtered = []
        for p in payments:
            pd_ = str(p.get("payment_date", ""))[:10]
            if sd <= pd_ <= ed:
                filtered.append(p)

        daily = {}
        methods = {}
        for p in filtered:
            d = str(p.get("payment_date", ""))[:10]
            daily[d] = daily.get(d, 0) + float(p.get("amount", 0) or 0)
            m = p.get("payment_method", "Other")
            methods[m] = methods.get(m, 0) + float(p.get("amount", 0) or 0)

        lines = [f"PAYMENT ANALYSIS — {sd} to {ed}", "=" * 60, ""]
        lines.append("Daily Collections:")
        for d in sorted(daily):
            lines.append(f"  • {d}: ₹{_fmt_inr(daily[d])}")
        lines.append("")
        lines.append("By Method:")
        for m, amt in methods.items():
            lines.append(f"  • {m}: ₹{_fmt_inr(amt)}")
        return "\n".join(lines)

    def _load_photo_list(self):
        try:
            self._photo_list.controls.clear()
            travelers = self.db.get_travelers()
            travelers.sort(key=lambda t: (
                f"{t.get('first_name','')} {t.get('last_name','')}".lower()))
            self._photo_cache = travelers
            for t in travelers:
                self._photo_list.controls.append(
                    ft.TextButton(
                        content=ft.Text(
                            f"{t.get('first_name','')} "
                            f"{t.get('last_name','')}"),
                        on_click=lambda e, tt=t: self._show_photo(tt),
                    ))
            self.page.update()
        except Exception as ex:
            print(f"_load_photo_list: {ex}")

    def _filter_photo_list(self, e=None):
        q = (self._photo_search.value or "").lower()
        self._photo_list.controls.clear()
        for t in self._photo_cache:
            name = (f"{t.get('first_name','')} "
                    f"{t.get('last_name','')}").lower()
            if q in name:
                self._photo_list.controls.append(
                    ft.TextButton(
                        content=ft.Text(name.title()),
                        on_click=lambda e, tt=t: self._show_photo(tt),
                    ))
        self.page.update()

    def _show_photo(self, traveler):
        base = _app_base()
        rel = traveler.get("photo", "")
        path = ""
        if rel and os.path.exists(os.path.join(base, rel)):
            path = os.path.join(base, rel)
        else:
            tid = traveler.get("id", "")
            if tid:
                folder = tid.replace("/", "_").replace("\\", "_")
                pdir = os.path.join(base, "documents", folder, "photos")
                if os.path.exists(pdir):
                    for f in os.listdir(pdir):
                        if f.lower().endswith((".jpg", ".jpeg", ".png",
                                                ".bmp", ".gif")):
                            path = os.path.join(pdir, f)
                            break

        if path and os.path.exists(path):
            self._photo_display.content = ft.Column([
                ft.Image(src=path, fit=ft.BoxFit.CONTAIN, height=400),
                ft.Text(os.path.basename(path), size=11,
                        color=ft.Colors.GREY_600),
            ], horizontal_alignment=ft.CrossAxisAlignment.CENTER)
        else:
            self._photo_display.content = ft.Text(
                "📷  No photo available", color=ft.Colors.GREY_500)
        self.page.update()
